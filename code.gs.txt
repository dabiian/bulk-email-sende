function enviarCorreosMasivosConAdjuntos() {
  const hoja = SpreadsheetApp.getActiveSpreadsheet().getSheetByName("Registro de Estudiantes")
  const rango = hoja.getDataRange();

  for (i = 2; i <= rango.getLastRow(); i++) {
    const email = hoja.getRange(i, 1).getValue()
    const email2 = hoja.getRange(i, 2).getValue()
    const resultado = hoja.getRange(i, 5).getValue()
    const confirmacion = hoja.getRange(i, 6).getValue()
    var idArchivo = hoja.getRange(i, 4).getValue()
    const archivo = DriveApp.getFileById(idArchivo)
    const blob = archivo.getAs(MimeType.PDF)
    
    var textoHtml1=HtmlService.createHtmlOutputFromFile("Aprobo").getContent();
    if (confirmacion == "" && resultado == "Aprobo" && archivo != "") {
      MailApp.sendEmail(email, "Resultado examen OOPT","",{htmlBody:textoHtml1,attachments: [blob]})
      hoja.getRange(i, 6).setValue("Enviado")
      }
    if (confirmacion == "" && resultado == "Aprobo" && archivo != "") {
      MailApp.sendEmail(email2, "Resultado examen OOPT","",{htmlBody:textoHtml1,attachments: [blob]})
      hoja.getRange(i, 6).setValue("Enviado")
      }
   
   var textoHtml2=HtmlService.createHtmlOutputFromFile("No aprobo").getContent();
   if (confirmacion == "" && resultado == "No aprobo" && archivo != "") {
      MailApp.sendEmail(email, "Resultado examen OOPT","",{htmlBody:textoHtml2,attachments: [blob]})
      hoja.getRange(i, 6).setValue("Enviado")
      }
   if (confirmacion == "" && resultado == "No aprobo" && archivo != "") {
      MailApp.sendEmail(email2, "Resultado examen OOPT","",{htmlBody:textoHtml2,attachments: [blob]})
      hoja.getRange(i, 6).setValue("Enviado")
      }
  }

}
